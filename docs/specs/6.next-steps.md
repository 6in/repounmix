### ステップ2: 基本レイアウト実装

#### 2-1. App.tsx（メインアプリケーション）

**ファイル**: `src/App.tsx`

**実装内容**:
```typescript
import { useState } from 'react';
import { RepomixCore } from './services/RepomixCore';
import { Toolbar } from './components/Toolbar/Toolbar';
import { FileTree } from './components/FileTree/FileTree';
import { EditorTabs } from './components/Editor/EditorTabs';
import './App.css';

function App() {
  const [core] = useState(() => new RepomixCore());
  const [selectedFile, setSelectedFile] = useState<string | null>(null);
  const [activeTab, setActiveTab] = useState<'file-editor' | 'repomix-xml'>('file-editor');

  const handleOpenFolder = async () => {
    try {
      await core.openFolder();
      // フォルダツリー更新
    } catch (error) {
      console.error('Failed to open folder:', error);
    }
  };

  const handleMix = () => {
    const xml = core.mix();
    setActiveTab('repomix-xml');
    // XMLをエディタに表示
  };

  const handleUnmix = async (xml: string) => {
    try {
      await core.unmix(xml);
      // フォルダツリー更新
    } catch (error) {
      console.error('Failed to unmix:', error);
    }
  };

  return (
    <div className="app">
      <Toolbar
        onOpenFolder={handleOpenFolder}
        onMix={handleMix}
        onUnmix={handleUnmix}
      />
      <div className="main-content">
        <FileTree
          fileManager={core.getFileManager()}
          selectedFile={selectedFile}
          onFileSelect={(path) => {
            setSelectedFile(path);
            setActiveTab('file-editor');
          }}
        />
        <EditorTabs
          activeTab={activeTab}
          onTabChange={setActiveTab}
          selectedFile={selectedFile}
          fileManager={core.getFileManager()}
        />
      </div>
    </div>
  );
}

export default App;
```

**ファイル**: `src/components/FileTree/FileTreeNode.tsx`
```typescript
import { DirectoryNode, FileNode } from '../../types';

interface FileTreeNodeProps {
  node: DirectoryNode | FileNode;
  selectedFile: string | null;
  onFileSelect: (path: string) => void;
}

export function FileTreeNode({ node, selectedFile, onFileSelect }: FileTreeNodeProps) {
  if ('isDirectory' in node && node.isDirectory) {
    return (
      <div className="directory-node">
        <div className="directory-name">📁 {node.name}</div>
        <div className="directory-children">
          {node.children.map((child, index) => (
            <FileTreeNode
              key={index}
              node={child}
              selectedFile={selectedFile}
              onFileSelect={onFileSelect}
            />
          ))}
        </div>
      </div>
    );
  }

  const fileNode = node as FileNode;
  const isSelected = fileNode.path === selectedFile;

  return (
    <div
      className={`file-node ${isSelected ? 'selected' : ''}`}
      onClick={() => onFileSelect(fileNode.path)}
    >
      📄 {fileNode.path.split('/').pop()}
    </div>
  );
}
```

#### 2-4. EditorTabsコンポーネント

**ファイル**: `src/components/Editor/EditorTabs.tsx`

**実装内容**:
```typescript
import Editor from '@monaco-editor/react';
import { FileManager } from '../../core/FileManager';

interface EditorTabsProps {
  activeTab: 'file-editor' | 'repomix-xml';
  onTabChange: (tab: 'file-editor' | 'repomix-xml') => void;
  selectedFile: string | null;
  fileManager: FileManager;
}

export function EditorTabs({
  activeTab,
  onTabChange,
  selectedFile,
  fileManager
}: EditorTabsProps) {
  const file = selectedFile ? fileManager.getFile(selectedFile) : null;

  return (
    <div className="editor-tabs">
      <div className="tabs">
        <button
          className={activeTab === 'file-editor' ? 'active' : ''}
          onClick={() => onTabChange('file-editor')}
        >
          File Editor
        </button>
        <button
          className={activeTab === 'repomix-xml' ? 'active' : ''}
          onClick={() => onTabChange('repomix-xml')}
        >
          Repomix XML
        </button>
      </div>
      <div className="editor-content">
        {activeTab === 'file-editor' && (
          <Editor
            height="100%"
            defaultLanguage="typescript"
            value={file?.content || '// ファイルを選択してください'}
            onChange={(value) => {
              if (file && value !== undefined) {
                fileManager.setFile({ ...file, content: value });
              }
            }}
          />
        )}
        {activeTab === 'repomix-xml' && (
          <Editor
            height="100%"
            defaultLanguage="xml"
            value="<!-- Repomix XMLをここに貼り付けてください -->"
          />
        )}
      </div>
    </div>
  );
}
```

## 開発時の注意事項

### 1. コア機能は変更しない
Phase 1で完成したコア機能（`src/core/`, `src/adapters/`, `src/services/`）は変更しないでください。すべてのテストが成功しているため、変更するとバグの原因になります。

### 2. Monaco Editorの設定
```typescript
import Editor from '@monaco-editor/react';

// 推奨設定
<Editor
  height="100%"
  theme="vs-dark"  // or "vs-light"
  options={{
    minimap: { enabled: false },
    fontSize: 14,
    wordWrap: 'on'
  }}
/>
```

### 3. File System Access APIの権限
ユーザーがフォルダを選択するたびに権限を要求する必要があります。権限が失効した場合は再要求してください。

### 4. エラーハンドリング
すべての非同期操作は`try-catch`で囲んでください。

## トラブルシューティング

### 問題: テストが失敗する
**解決策**:
```bash
# キャッシュをクリア
rm -rf node_modules
npm install

# テスト再実行
npm test
```

### 問題: Monaco Editorが表示されない
**解決策**:
```bash
# monaco-editorを再インストール
npm uninstall monaco-editor @monaco-editor/react
npm install monaco-editor @monaco-editor/react
```

### 問題: File System Access APIが動作しない
**解決策**:
- Chrome/Edgeを使用してください
- HTTPSまたはlocalhostで実行してください
- ブラウザの権限設定を確認してください
